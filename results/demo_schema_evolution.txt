-- 1) ADD COLUMN: новая колонка-сегмент клиента. Меняются только метаданные (новый schema-id), parquet-файлы НЕ переписываются; в старых файлах колонки нет -> читается как NULL

trino> ALTER TABLE silver.lakehouse.dim_customer ADD COLUMN customer_tier varchar;

trino> SELECT custkey, acctbal, customer_tier FROM silver.lakehouse.dim_customer ORDER BY custkey LIMIT 3;
custkey | acctbal | customer_tier
1 | 711.56 | None
2 | 121.65 | None
3 | 7498.12 | None

trino> UPDATE silver.lakehouse.dim_customer SET customer_tier = CASE WHEN acctbal >= 5000 THEN 'GOLD' WHEN acctbal >= 0 THEN 'STANDARD' ELSE 'DEBTOR' END;
rows
7500000

-- 2) RENAME COLUMN: Iceberg адресует колонки по field-id, а не по имени -> переименование мгновенное, данные не трогаются
